data = raw.get_data()

variances = np.var(data, axis=1)

threshold_high = np.mean(variances) + 3*np.std(variances)
threshold_low = np.mean(variances) - 3*np.std(variances)

bad_var = np.where((variances > threshold_high) | (variances < threshold_low))[0]
bad_channels = [raw.ch_names[i] for i in bad_var]

print("Bad (varianza):", bad_channels)
